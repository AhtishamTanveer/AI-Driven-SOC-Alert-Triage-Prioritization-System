def classify_priority(score):
    if score >= 100:
        return 'CRITICAL'     # 100+ points
    elif score >= 70:
        return 'HIGH'         # 70-99 points
    elif score >= 40:
        return 'MEDIUM'       # 40-69 points
    else:
        return 'LOW'          # 0-39 points
```
